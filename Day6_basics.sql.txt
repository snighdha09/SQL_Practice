Q1-
Select emp_name,salary,
Case
When salary>=80000 then 'High'
When salary between 50000 and 90000 then 'Medium'
Else 'low'
END As salary_category
From Employees;

Q2-
Select emp_name,
Case
When dept_id=10 then salary *0.20
When dept_id=20 then salary *0.15
Else salary*0.10
END As bonus
From Employees;

Q3-

Select city, count(*) As total_employee,SUM(Case When salary>70000 then 1 else 0 end) As high_salary_count
From Employees group by city;

Q4-
Select dept_id, Sum(salary) As total_salary, Count(*) As total_employees, Avg(salary) AS avg_salary, SUM(CASE When city='Delhi' then 1 else 0 end) AS Count_employees_From _delhi From Employees group by dept_id;

Q5-
Select emp_name, coalesce(city,'unknown') As city from Employees;

Q6-
Select emp_name, coalesce(salary,0) As salary from Employees;

Q7-

Q8-

Q9-
Select Count(*) As total_employess, Extract(Month from Joining_date) As Month_no From Orders where Extract(Year from Joining_date) As Year_no=2024 group by Month_no Order by Month_no;

Q10-
Select count(order_id) As total_orders, Extract(Month from Order_date) As Month_no, Sum(Amount) As total_amount From Orders where Extract(Year from Order_date) As Year_no=2024 group by Month_no Order by Month_no;

Q11-
Select customer_id,sum(amount) As total_amount From Orders Where Status='Delivered' group by customer_id;

Q12-
Select Count(CASE When Status='Delivered' then 1 else 0 END) AS Delivered_count,Count(CASE When Status='Cancelled' then 1 else 0 END) AS Cancelled_count,Count(CASE When Status='Returned' then 1 else 0 END) AS Returned_count From Orders;

Q13-
